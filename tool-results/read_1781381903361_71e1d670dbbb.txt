   230→   230→    .toLowerCase()

   233→   233→    .replace(/[\s_]+/g, "-")

   235→   235→    .replace(/^-+|-+$/g, "");
   236→   236→}

   242→   242→      year: "numeric",

   245→   245→    });

   251→   251→// ─── Toast System ───────────────────────────────────────────────────────────────

   253→   253→interface ToastMessage {
   254→   254→  id: number;
   255→   255→  message: string;
   256→   256→  type: "success" | "error";
   257→   257→}
   258→   258→
   259→   259→let toastIdCounter = 0;
   260→   260→
   261→   261→function useToast() {
   262→   262→  const [toasts, setToasts] = useState<ToastMessage[]>([]);
   263→   263→
   264→   264→  const showToast = useCallback((message: string, type: "success" | "error" = "success") => {
   265→   265→    const id = ++toastIdCounter;
   266→   266→    setToasts((prev) => [...prev, { id, message, type }]);
   267→   267→    setTimeout(() => {
   268→   268→      setToasts((prev) => prev.filter((t) => t.id !== id));
   269→   269→    }, 3500);
   270→   270→  }, []);
   271→   271→
   272→   272→  const ToastContainer = (
   273→   273→    <div className="fixed top-4 right-4 z-[100] flex flex-col gap-2">
   274→   274→      {toasts.map((toast) => (
   275→   275→        <div
   276→   276→          key={toast.id}
   277→   277→          className={`px-4 py-3 rounded-xl shadow-lg text-sm font-[family-name:var(--font-inter)] flex items-center gap-2 animate-in slide-in-from-right duration-300 ${
   278→   278→            toast.type === "success"
   279→   279→              ? "bg-green-500/90 text-white backdrop-blur-sm"
   280→   280→              : "bg-red-500/90 text-white backdrop-blur-sm"
   281→   281→          }`}
   282→   282→        >
   283→   283→          {toast.type === "success" ? (
   284→   284→            <CheckCircle className="h-4 w-4 shrink-0" />
   285→   285→          ) : (
   286→   286→            <AlertCircle className="h-4 w-4 shrink-0" />
   287→   287→          )}
   288→   288→          {toast.message}
   289→   289→        </div>
   290→   290→      ))}
   291→   291→    </div>
   292→   292→  );
   293→   293→
   294→   294→  return { showToast, ToastContainer };
   295→   295→}
