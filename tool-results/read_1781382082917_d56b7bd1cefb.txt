   228→function generateSlug(text: string): string {
   229→  return text
   230→    .toLowerCase()
   231→    .trim()
   232→    .replace(/[^\w\s-]/g, "")
   233→    .replace(/[\s_]+/g, "-")
   234→    .replace(/-+/g, "-")
   235→    .replace(/^-+|-+$/g, "");
   236→}

   238→function formatDate(dateStr: string): string {
   239→  if (!dateStr) return "—";
   240→  try {
   241→    return new Date(dateStr).toLocaleDateString("en-US", {
   242→      year: "numeric",
   243→      month: "short",
   244→      day: "numeric",
   245→    });
   246→  } catch {
   247→    return "—";
   248→  }
   249→}

   251→// ─── Toast System ───────────────────────────────────────────────────────────────
   252→
   253→interface ToastMessage {
   254→  id: number;
   255→  message: string;
   256→  type: "success" | "error";
   257→}
   258→
   259→let toastIdCounter = 0;
   260→
   261→function useToast() {
   262→  const [toasts, setToasts] = useState<ToastMessage[]>([]);
   263→
   264→  const showToast = useCallback((message: string, type: "success" | "error" = "success") => {
   265→    const id = ++toastIdCounter;
   266→    setToasts((prev) => [...prev, { id, message, type }]);
   267→    setTimeout(() => {
   268→      setToasts((prev) => prev.filter((t) => t.id !== id));
   269→    }, 3500);
   270→  }, []);
   271→
   272→  const ToastContainer = (
   273→    <div className="fixed top-4 right-4 z-[100] flex flex-col gap-2">
   274→      {toasts.map((toast) => (
   275→        <div
   276→          key={toast.id}
   277→          className={`px-4 py-3 rounded-xl shadow-lg text-sm font-[family-name:var(--font-inter)] flex items-center gap-2 animate-in slide-in-from-right duration-300 ${
   278→            toast.type === "success"
   279→              ? "bg-green-500/90 text-white backdrop-blur-sm"
   280→              : "bg-red-500/90 text-white backdrop-blur-sm"
   281→          }`}
   282→        >
   283→          {toast.type === "success" ? (
   284→            <CheckCircle className="h-4 w-4 shrink-0" />
   285→          ) : (
   286→            <AlertCircle className="h-4 w-4 shrink-0" />
   287→          )}
   288→          {toast.message}
   289→        </div>
   290→      ))}
   291→    </div>
   292→  );
   293→
   294→  return { showToast, ToastContainer };
   295→}

   525→export default function AdminDashboard() {
   526→  // Auth state
   527→  const [isAuthenticated, setIsAuthenticated] = useState(false);
   528→  const [authUser, setAuthUser] = useState<{ username: string; role: string } | null>(null);
   529→  const [loginForm, setLoginForm] = useState({ username: "", password: "" });
   530→  const [loginError, setLoginError] = useState("");
   531→  const [loginLoading, setLoginLoading] = useState(false);
   532→
   533→  // Check for existing session on mount
   534→  useEffect(() => {
   535→    const session = sessionStorage.getItem("clipe233_admin");
   536→    if (session) {
   537→      try {
   538→        const data = JSON.parse(session);
   539→        setIsAuthenticated(true);
   540→        setAuthUser(data.user);
   541→      } catch {
   542→        sessionStorage.removeItem("clipe233_admin");
   543→      }
   544→    }
   545→  }, []);
   546→
   547→  const handleLogin = async (e: React.FormEvent) => {
   548→    e.preventDefault();
   549→    setLoginError("");
   550→    setLoginLoading(true);
   551→    try {
   552→      const res = await fetch("/api/auth", {
   553→        method: "POST",
   554→        headers: { "Content-Type": "application/json" },
   555→        body: JSON.stringify(loginForm),
   556→      });
   557→      const data = await res.json();
   558→      if (!res.ok) {
   559→        setLoginError(data.error || "Invalid credentials");
   560→        return;
   561→      }
   562→      sessionStorage.setItem("clipe233_admin", JSON.stringify(data));
   563→      setIsAuthenticated(true);
   564→      setAuthUser(data.user);
   565→    } catch {
   566→      setLoginError("Connection error. Please try again.");
   567→    } finally {
   568→      setLoginLoading(false);
   569→    }
   570→  };
   571→
   572→  const handleLogout = () => {
   573→    sessionStorage.removeItem("clipe233_admin");
   574→    setIsAuthenticated(false);
   575→    setAuthUser(null);
   576→  };
   577→
   578→  // Login screen
   579→  if (!isAuthenticated) {
   580→    return (
   581→      <div className="min-h-screen bg-black flex items-center justify-center p-4">
   582→        <div className="w-full max-w-md">
   583→          {/* Logo */}
   584→          <div className="text-center mb-8">
   585→            <div className="w-16 h-16 rounded-2xl bg-falu/20 flex items-center justify-center mx-auto mb-4">
   586→              <ShieldCheck className="h-8 w-8 text-falu-light" />
   587→            </div>
   588→            <h1 className="text-2xl font-bold font-[family-name:var(--font-poppins)] text-white">Clipe233 Admin</h1>
   589→            <p className="text-silver/50 font-[family-name:var(--font-inter)] text-sm mt-1">Sign in to manage your website</p>
   590→          </div>
   591→
   592→          {/* Login Form */}
   593→          <div className="glass-card rounded-2xl p-8">
   594→            <form onSubmit={handleLogin} className="space-y-5">
   595→              {loginError && (
   596→                <div className="p-3 rounded-lg bg-red-900/30 border border-red-800/50 flex items-center gap-2">
   597→                  <AlertCircle className="h-4 w-4 text-red-400 flex-shrink-0" />
   598→                  <span className="text-red-300 text-sm font-[family-name:var(--font-inter)]">{loginError}</span>
   599→                </div>
   600→              )}
   601→              <div>
   602→                <Label className="text-silver/70 font-[family-name:var(--font-inter)] mb-1.5 block">Username</Label>
   603→                <div className="relative">
   604→                  <Input
   605→                    value={loginForm.username}
   606→                    onChange={(e) => setLoginForm((p) => ({ ...p, username: e.target.value }))}
   607→                    placeholder="Enter username"
   608→                    className="bg-white/5 border-white/10 text-white placeholder:text-silver/30 font-[family-name:var(--font-inter)] pl-10"
   609→                    required
   610→                  />
   611→                  <Users className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-silver/30" />
   612→                </div>
   613→              </div>
   614→              <div>
   615→                <Label className="text-silver/70 font-[family-name:var(--font-inter)] mb-1.5 block">Password</Label>
   616→                <div className="relative">
   617→                  <Input
   618→                    type="password"
   619→                    value={loginForm.password}
   620→                    onChange={(e) => setLoginForm((p) => ({ ...p, password: e.target.value }))}
   621→                    placeholder="Enter password"
   622→                    className="bg-white/5 border-white/10 text-white placeholder:text-silver/30 font-[family-name:var(--font-inter)] pl-10"
   623→                    required
   624→                  />
   625→                  <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-silver/30" />
   626→                </div>
   627→              </div>
   628→              <Button
   629→                type="submit"
   630→                disabled={loginLoading}
   631→                className="w-full bg-falu hover:bg-falu-light text-white glow-red-sm hover:glow-red transition-all duration-300 font-[family-name:var(--font-inter)]"

  1078→function DashboardView({
  1079→  contacts,
  1080→  blogPosts,
  1081→  projects,
  1082→  teamMembers,
  1083→  services,
  1084→  subscribers,
  1085→  jobs,
  1086→  testimonials,
  1087→  apiHealth,
  1088→  settings,
  1089→  onNavigate,
  1090→}: {
  1091→  contacts: ContactItem[];
  1092→  blogPosts: BlogPost[];
  1093→  projects: Project[];
  1094→  teamMembers: TeamMember[];
  1095→  services: ServiceItem[];
  1096→  subscribers: NewsletterSubscriber[];
  1097→  jobs: JobOpening[];
  1098→  testimonials: Testimonial[];
  1099→  apiHealth: Record<string, string> | null;
  1100→  settings: SiteSetting[];
  1101→  onNavigate: (section: string) => void;
  1102→}) {
  1103→  const oneWeekAgo = new Date();
  1104→  oneWeekAgo.setDate(oneWeekAgo.getDate() - 7);
  1105→
  1106→  const oneMonthAgo = new Date();
  1107→  oneMonthAgo.setDate(oneMonthAgo.getDate() - 30);
  1108→
  1109→  const newContactsWeek = contacts.filter((c) => new Date(c.createdAt) >= oneWeekAgo).length;
  1110→  const newContactsMonth = contacts.filter((c) => new Date(c.createdAt) >= oneMonthAgo).length;
  1111→  const closedContacts = contacts.filter((c) => c.status === "closed").length;
  1112→  const conversionRate = contacts.length > 0 ? Math.round((closedContacts / contacts.length) * 100) : 0;
  1113→
  1114→  // Last 7 days contacts chart data
  1115→  const last7Days = Array.from({ length: 7 }, (_, i) => {
  1116→    const d = new Date();
  1117→    d.setDate(d.getDate() - (6 - i));
  1118→    d.setHours(0, 0, 0, 0);
  1119→    return d;
  1120→  });
  1121→  const contactsPerDay = last7Days.map((day) => {
  1122→    const nextDay = new Date(day);
  1123→    nextDay.setDate(nextDay.getDate() + 1);
  1124→    return contacts.filter((c) => {
  1125→      const cd = new Date(c.createdAt);
  1126→      return cd >= day && cd < nextDay;
  1127→    }).length;
  1128→  });
  1129→  const maxContactsPerDay = Math.max(...contactsPerDay, 1);
  1130→
  1131→  // Contact status breakdown
  1132→  const statusBreakdown = [
  1133→    { label: "New", count: contacts.filter((c) => c.status === "new").length, color: "bg-blue-400" },
  1134→    { label: "Contacted", count: contacts.filter((c) => c.status === "contacted").length, color: "bg-yellow-400" },
  1135→    { label: "Qualified", count: contacts.filter((c) => c.status === "qualified").length, color: "bg-green-400" },
  1136→    { label: "Closed", count: contacts.filter((c) => c.status === "closed").length, color: "bg-silver/40" },
  1137→  ];
  1138→
  1139→  // Top services by inquiries
  1140→  const serviceInquiries: Record<string, number> = {};
  1141→  contacts.forEach((c) => {
  1142→    if (c.service) {
  1143→      serviceInquiries[c.service] = (serviceInquiries[c.service] || 0) + 1;
  1144→    }
  1145→  });
  1146→  const topServices = Object.entries(serviceInquiries)
  1147→    .sort(([, a], [, b]) => b - a)
  1148→    .slice(0, 5);
