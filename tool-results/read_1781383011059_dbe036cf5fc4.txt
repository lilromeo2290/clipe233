   228→function generateSlug(text: string): string {
   229→  return text
   230→    .toLowerCase()
   231→    .trim()
   232→    .replace(/[^\w\s-]/g, "")
   233→    .replace(/[\s_]+/g, "-")
   234→    .replace(/-+/g, "-")
   235→    .replace(/^-+|-+$/g, "");
   236→}

   238→function formatDate(dateStr: string): string {
   239→  if (!dateStr) return "—";
   240→  try {
   241→    return new Date(dateStr).toLocaleDateString("en-US", {
   242→      year: "numeric",
   243→      month: "short",
   244→      day: "numeric",
   245→    });
   246→  } catch {
   247→    return "—";
   248→  }
   249→}

   251→// ─── Toast System ───────────────────────────────────────────────────────────────
   252→
   253→interface ToastMessage {
   254→  id: number;
   255→  message: string;
   256→  type: "success" | "error";
   257→}
   258→
   259→let toastIdCounter = 0;
   260→
   261→function useToast() {
   262→  const [toasts, setToasts] = useState<ToastMessage[]>([]);
   263→
   264→  const showToast = useCallback((message: string, type: "success" | "error" = "success") => {
   265→    const id = ++toastIdCounter;
   266→    setToasts((prev) => [...prev, { id, message, type }]);
   267→    setTimeout(() => {
   268→      setToasts((prev) => prev.filter((t) => t.id !== id));
   269→    }, 3500);
   270→  }, []);
   271→
   272→  const ToastContainer = (
   273→    <div className="fixed top-4 right-4 z-[100] flex flex-col gap-2">
   274→      {toasts.map((toast) => (
   275→        <div
   276→          key={toast.id}
   277→          className={`px-4 py-3 rounded-xl shadow-lg text-sm font-[family-name:var(--font-inter)] flex items-center gap-2 animate-in slide-in-from-right duration-300 ${
   278→            toast.type === "success"
   279→              ? "bg-green-500/90 text-white backdrop-blur-sm"
   280→              : "bg-red-500/90 text-white backdrop-blur-sm"
   281→          }`}
   282→        >
   283→          {toast.type === "success" ? (
   284→            <CheckCircle className="h-4 w-4 shrink-0" />
   285→          ) : (
   286→            <AlertCircle className="h-4 w-4 shrink-0" />
   287→          )}
   288→          {toast.message}
   289→        </div>
   290→      ))}
   291→    </div>
   292→  );
   293→
   294→  return { showToast, ToastContainer };
   295→}

   523→// ─── Login Component ────────────────────────────────────────────────────────────
   524→
   525→function AdminLogin({ onLogin }: { onLogin: (user: { username: string; role: string }) => void }) {
   526→  const [loginForm, setLoginForm] = useState({ username: "", password: "" });
   527→  const [loginError, setLoginError] = useState("");
   528→  const [loginLoading, setLoginLoading] = useState(false);
   529→
   530→  const handleLogin = async (e: React.FormEvent) => {
   531→    e.preventDefault();
   532→    setLoginError("");
   533→    setLoginLoading(true);
   534→    try {
   535→      const res = await fetch("/api/auth", {
   536→        method: "POST",
   537→        headers: { "Content-Type": "application/json" },
   538→        body: JSON.stringify(loginForm),
   539→      });
   540→      const data = await res.json();
   541→      if (!res.ok) {
   542→        setLoginError(data.error || "Invalid credentials");
   543→        return;
   544→      }
   545→      sessionStorage.setItem("clipe233_admin", JSON.stringify(data));
   546→      onLogin(data.user);
   547→    } catch {
   548→      setLoginError("Connection error. Please try again.");
   549→    } finally {
   550→      setLoginLoading(false);
   551→    }
   552→  };
   553→
   554→  return (
   555→    <div className="min-h-screen bg-black flex items-center justify-center p-4">
   556→      <div className="w-full max-w-md">
   557→        {/* Logo */}
   558→        <div className="text-center mb-8">
   559→          <div className="w-16 h-16 rounded-2xl bg-falu/20 flex items-center justify-center mx-auto mb-4">
   560→            <ShieldCheck className="h-8 w-8 text-falu-light" />
   561→          </div>
   562→          <h1 className="text-2xl font-bold font-[family-name:var(--font-poppins)] text-white">Clipe233 Admin</h1>
   563→          <p className="text-silver/50 font-[family-name:var(--font-inter)] text-sm mt-1">Sign in to manage your website</p>
   564→        </div>
   565→
   566→        {/* Login Form */}
   567→        <div className="glass-card rounded-2xl p-8">
   568→          <form onSubmit={handleLogin} className="space-y-5">
   569→            {loginError && (
   570→              <div className="p-3 rounded-lg bg-red-900/30 border border-red-800/50 flex items-center gap-2">
   571→                <AlertCircle className="h-4 w-4 text-red-400 flex-shrink-0" />
   572→                <span className="text-red-300 text-sm font-[family-name:var(--font-inter)]">{loginError}</span>
   573→              </div>
   574→            )}
   575→            <div>
   576→              <Label className="text-silver/70 font-[family-name:var(--font-inter)] mb-1.5 block">Username</Label>
   577→              <div className="relative">
   578→                <Input
   579→                  value={loginForm.username}
   580→                  onChange={(e) => setLoginForm((p) => ({ ...p, username: e.target.value }))}
   581→                  placeholder="Enter username"
   582→                  className="bg-white/5 border-white/10 text-white placeholder:text-silver/30 font-[family-name:var(--font-inter)] pl-10"
   583→                  required
   584→                />
   585→                <Users className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-silver/30" />
   586→              </div>
   587→            </div>
   588→            <div>
   589→              <Label className="text-silver/70 font-[family-name:var(--font-inter)] mb-1.5 block">Password</Label>
   590→              <div className="relative">
   591→                <Input
   592→                  type="password"
   593→                  value={loginForm.password}
   594→                  onChange={(e) => setLoginForm((p) => ({ ...p, password: e.target.value }))}
   595→                  placeholder="Enter password"
   596→                  className="bg-white/5 border-white/10 text-white placeholder:text-silver/30 font-[family-name:var(--font-inter)] pl-10"
   597→                  required
   598→                />
   599→                <Lock className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-silver/30" />
   600→              </div>
   601→            </div>
   602→            <Button
   603→              type="submit"
   604→              disabled={loginLoading}
   605→              className="w-full bg-falu hover:bg-falu-light text-white glow-red-sm hover:glow-red transition-all duration-300 font-[family-name:var(--font-inter)]"
   606→            >
   607→              {loginLoading ? (
   608→                <><Loader2 className="mr-2 h-4 w-4 animate-spin" /> Signing in...</>
   609→              ) : (
   610→                <><Lock className="mr-2 h-4 w-4" /> Sign In</>
   611→              )}
   612→            </Button>
   613→          </form>
   614→        </div>
   615→
   616→        <p className="text-center text-silver/30 text-xs font-[family-name:var(--font-inter)] mt-6">
   617→          Clipe233 Engineers &copy; {new Date().getFullYear()} — Admin CMS
   618→        </p>
   619→      </div>
   620→    </div>
   621→  );
   622→}

  1096→function DashboardView({

  1120→}) {
  1121→  const oneWeekAgo = new Date();
  1122→  oneWeekAgo.setDate(oneWeekAgo.getDate() - 7);
  1123→
  1124→  const oneMonthAgo = new Date();
  1125→  oneMonthAgo.setDate(oneMonthAgo.getDate() - 30);
  1126→
  1127→  const newContactsWeek = contacts.filter((c) => new Date(c.createdAt) >= oneWeekAgo).length;
  1128→  const newContactsMonth = contacts.filter((c) => new Date(c.createdAt) >= oneMonthAgo).length;
  1129→  const closedContacts = contacts.filter((c) => c.status === "closed").length;
  1130→  const conversionRate = contacts.length > 0 ? Math.round((closedContacts / contacts.length) * 100) : 0;
  1131→
  1132→  // Last 7 days contacts chart data
  1133→  const last7Days = Array.from({ length: 7 }, (_, i) => {
  1134→    const d = new Date();
  1135→    d.setDate(d.getDate() - (6 - i));
  1136→    d.setHours(0, 0, 0, 0);
  1137→    return d;
  1138→  });
  1139→  const contactsPerDay = last7Days.map((day) => {
  1140→    const nextDay = new Date(day);
  1141→    nextDay.setDate(nextDay.getDate() + 1);
  1142→    return contacts.filter((c) => {
  1143→      const cd = new Date(c.createdAt);
  1144→      return cd >= day && cd < nextDay;
  1145→    }).length;
  1146→  });
  1147→  const maxContactsPerDay = Math.max(...contactsPerDay, 1);
  1148→
  1149→  // Contact status breakdown
